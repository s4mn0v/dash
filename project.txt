let RAW = [],
  filtered = [],
  chart = null;

async function init() {
  const res = await fetch("/api/data");
  RAW = await res.json();
  applyFilters();
  setupNav();
}

function setupNav() {
  document.querySelectorAll(".nav-item").forEach((btn) => {
    btn.onclick = (e) => {
      document
        .querySelectorAll(".nav-item, .view")
        .forEach((el) => el.classList.remove("active"));
      const view = e.target.dataset.view;
      e.target.classList.add("active");
      document.getElementById("v-" + view).classList.add("active");
      document.getElementById("view-title").textContent = e.target.textContent;
    };
  });
}

function applyFilters() {
  const marca = document.getElementById("f-marca").value;
  const q = document.getElementById("search").value.toLowerCase();

  filtered = RAW.filter((d) => {
    return (
      (!marca || d.marca === marca) &&
      (!q || d.mop.toLowerCase().includes(q) || d.ref.toLowerCase().includes(q))
    );
  });

  renderKPIs();
  renderTables();
  renderChart();
}

function renderKPIs() {
  const plan = filtered.reduce((s, d) => s + d.plan, 0);
  const comp = filtered.reduce((s, d) => s + d.comp, 0);
  const html = `
        <div class="kpi"><h3>MOPs</h3><p>${filtered.length}</p></div>
        <div class="kpi"><h3>Planeado</h3><p>${Math.round(plan).toLocaleString()}</p></div>
        <div class="kpi"><h3>Cortado</h3><p>${Math.round(comp).toLocaleString()}</p></div>
        <div class="kpi"><h3>% Efic.</h3><p>${plan ? Math.round((comp / plan) * 100) : 0}%</p></div>
    `;
  document.getElementById("kpis").innerHTML = html;
}

function renderTables() {
  // Entrega Real Table
  const entrega = RAW.filter((d) => d.section === "entrega");
  const entBody = document.querySelector("#t-entrega tbody");
  entBody.innerHTML = filtered
    .slice(0, 20)
    .map(
      (d) => `
        <tr><td>${d.grupo_ent}</td><td>${d.mop}</td><td>${d.plan}</td><td>${d.comp}</td></tr>
    `,
    )
    .join("");

  // WIP Table
  const wipBody = document.querySelector("#t-wip tbody");
  wipBody.innerHTML = filtered
    .filter((d) => d.estado === "Aprobado")
    .slice(0, 20)
    .map(
      (d) => `
        <tr><td>${d.mop}</td><td>${d.ref}</td><td>PROCESO</td></tr>
    `,
    )
    .join("");
}

function renderChart() {
  const ctx = document.getElementById("chartUC").getContext("2d");
  if (chart) chart.destroy();

  const meses = [...new Set(filtered.map((d) => d.mes))].sort();
  chart = new Chart(ctx, {
    type: "bar",
    data: {
      labels: meses,
      datasets: [
        {
          label: "Unidades",
          data: meses.map((m) =>
            filtered.filter((d) => d.mes === m).reduce((s, d) => s + d.comp, 0),
          ),
          backgroundColor: "#1f56c8",
        },
      ],
    },
  });
}

async function upload() {
  const fileInput = document.getElementById("up-file");
  const sec = document.getElementById("up-sec").value;
  const files = fileInput.files;

  if (files.length === 0) return;

  const fd = new FormData();
  fd.append("section", sec);

  // Append all files to same key
  for (let i = 0; i < files.length; i++) {
    fd.append("files", files[i]);
  }

  const res = await fetch("/api/upload", {
    method: "POST",
    body: fd,
  });

  if (res.ok) {
    alert("CARGA EXITOSA");
    location.reload();
  }
}
init();

package main

import (
	"encoding/json"
	"io"
	"net/http"
	"os"
	"path/filepath"
	"strconv"

	"github.com/extrame/xls"
)

type Row struct {
	MOP string  `json:"mop"`
	Val float64 `json:"val"`
	Sec string  `json:"section"`
}

func uploadHandler(res http.ResponseWriter, req *http.Request) {
	req.ParseMultipartForm(50 << 20) // 50MB
	form := req.MultipartForm
	files := form.File["files"]
	sec := req.FormValue("section")

	for _, head := range files {
		file, _ := head.Open()
		path := filepath.Join("data", sec, head.Filename)
		out, _ := os.Create(path)
		io.Copy(out, file)
		file.Close()
		out.Close()
	}
	res.Write([]byte("ok"))
}

func dataHandler(res http.ResponseWriter, req *http.Request) {
	secs := []string{"entrega", "pending", "cut", "wip"}
	var all []Row
	for _, s := range secs {
		files, _ := filepath.Glob(filepath.Join("data", s, "*.xls"))
		for _, f := range files {
			xl, _ := xls.Open(f, "utf-8")
			sheet := xl.GetSheet(0)
			for i := 1; i <= int(sheet.MaxRow); i++ {
				r := sheet.Row(i)
				v, _ := strconv.ParseFloat(r.Col(8), 64)
				all = append(all, Row{MOP: r.Col(0), Val: v, Sec: s})
			}
		}
	}
	json.NewEncoder(res).Encode(all)
}

func main() {
	dirs := []string{"data/entrega", "data/pending", "data/cut", "data/wip"}
	for _, d := range dirs {
		os.MkdirAll(d, 0755)
	}

	http.HandleFunc("/api/upload", uploadHandler)
	http.HandleFunc("/api/data", dataHandler)
	http.Handle("/", http.FileServer(http.Dir("./public")))
	http.ListenAndServe(":8080", nil)
}
